# Fumox server pod, quadlet equivalent of the compose `server` service.
#
# The pod holds the public listener (:8080) and the admin panel (:8081)
# and NOTHING ELSE: the probe and the proxy kernel live in a second pod
# (fumox-worker.pod). The split is the point — the kernel handles
# untrusted proxy traffic and its configuration is writable by the probe,
# so it must not be able to reach the admin panel even inside a shared
# network namespace. The two pods exchange nothing over the network:
# the server and the probe talk through the shared SQLite volume, and
# the probe and the kernel through the shared config volume.
#
# Start:
#   systemctl --user daemon-reload
#   systemctl --user start fumox-pod.service        # server
#   systemctl --user start fumox-worker-pod.service  # probe + kernel
#
# Only the two ports below are published. The kernel's controller (9090)
# stays inside the worker pod and is never published, exactly like in
# compose.

[Unit]
Description=Fumox server pod (public endpoints + admin panel)
After=network-online.target
Wants=network-online.target

[Pod]
PodName=fumox

# 8080: public /sub + /src endpoints.
# 8081: admin panel, loopback-only like the compose mapping.
PublishPort=8080:8080
PublishPort=127.0.0.1:8081:8081

[Install]
WantedBy=default.target
