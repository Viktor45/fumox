# Fumox pod — quadlet equivalent of docker-compose.yml.
#
# One systemd-managed pod ("fumox") holding all three compose services in a
# single network namespace: fumox-server, fumox-probe and the meow-rs kernel.
# Starting the generated pod service brings up every .container unit that
# references this file:
#
#   systemctl --user daemon-reload
#   systemctl --user start fumox-pod.service
#
# Pod-internal wiring (containers in one pod share loopback — there is no
# per-container DNS like in compose):
#   probe -> meow REST API   127.0.0.1:9090   (see fumox-probe.container)
#   server + probe -> SQLite /app/data        (fumox-data volume)
#   probe + meow    -> config exchange /shared (fumox-shared volume)
#
# Only the two ports below are published; meow's external-controller (9090)
# and everything else stays pod-internal, exactly like in compose.

[Unit]
Description=Fumox pod (server + probe + meow)
After=network-online.target
Wants=network-online.target

[Pod]
PodName=fumox

# 8080: public /sub + /src endpoints.
# 8081: admin panel, loopback-only like the compose mapping.
PublishPort=8080:8080
PublishPort=127.0.0.1:8081:8081

[Install]
WantedBy=default.target
